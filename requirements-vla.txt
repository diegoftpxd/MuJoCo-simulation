# Dependencias de GPU para correr OpenVLA + LIBERO en el cluster.
#
# 1) torch (+ torchvision, que necesita timm) se instalan APARTE. La version
#    depende de la ARQUITECTURA de la GPU del cluster:
#      - GPU nueva (Blackwell, sm_100/sm_120, requiere CUDA 12.8+): torch >= 2.6
#        con el build de CUDA correcto, p. ej.:
#            pip install torch torchvision --index-url https://download.pytorch.org/whl/cu130
#      - GPU antigua (Pascal/Turing, 1080Ti/2080Ti): torch 2.2.0 + cu118 basta.
#
#    OJO con el checkpoint: desde torch 2.6 el default de torch.load es
#    weights_only=True, que rompe la carga de checkpoints de LIBERO/OpenVLA
#    ("UnpicklingError: Weights only load failed"). Si la GPU obliga a torch>=2.6,
#    aplica el shim (fuente confiable) ANTES de cargar nada:
#        import torch, functools
#        torch.load = functools.partial(torch.load, weights_only=False)
#    torch debe ser compatible con NumPy 1.x (ver numpy<2 abajo).
#
# 2) LIBERO (robosuite, bddl, ...) se instala desde el repo incluido:
#       pip install -r benchmarks/libero/Libero-10-r/requirements.txt
#       pip install -e benchmarks/libero/Libero-10-r
#
# 3) ESTE archivo se instala AL FINAL:  pip install -r requirements-vla.txt
#    LIBERO fija un transformers viejo (incompatible con el codigo remoto de
#    OpenVLA); instalar este archivo despues hace que sus pines tengan la ultima
#    palabra. Por eso transformers/tokenizers/huggingface_hub van FIJADOS.

# --- Stack OpenVLA (versiones fijadas por el codigo remoto del modelo) ---
transformers==4.40.1
tokenizers==0.19.1        # rango que exige transformers 4.40.1 (>=0.19,<0.20)
huggingface_hub==0.23.5   # transformers 4.40 se rompe con hub >=0.26
timm==0.9.10
accelerate>=0.25,<1.0     # device_map="auto"; accelerate 1.x no esta probado con transformers 4.40
hf_transfer

# --- Compatibilidad de LIBERO ---
# LIBERO usa la API de robosuite 1.4.x (clase SingleArmEnv). robosuite 1.5+ la
# elimino, asi que hay que fijar 1.4.x. Va aqui (se instala al final) para que
# gane sobre el robosuite mas nuevo que arrastra el requirements.txt de LIBERO.
robosuite==1.4.1

# --- Simulacion + render headless ---
# OJO: robosuite 1.4.1 (el que usa LIBERO) NO es compatible con mujoco 3.x; con
# 3.x falla con "'MjData' object has no attribute 'qM'". Hay que fijar 2.3.x.
# Esto DIFIERE a proposito de requirements.txt, que usa mujoco 3.x para la
# feature de objetos via MjSpec (no se usa en los experimentos de LIBERO).
mujoco==2.3.2
numpy<2                  # mujoco 2.3.2 y robosuite 1.4 no soportan NumPy 2.0
imageio
imageio-ffmpeg
